@override
ValidationResult<T> validate(String key, T input) {
  var ret = super.validate(key, input);

  if (ret.successful && (input < min || input > max)) {
    ret = ValidationResult._failure([
      'Value ($input) must be between $min and $max. (>= $min && <= $max)',
    ]);
  }

  return ret;
}
